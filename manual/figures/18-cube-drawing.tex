% The straight-line drawing of the cube that PlanarDrawing computes in the
% example 18-planar.cc (sec. 18.5): the eight nodes on the 6 x 6 grid at the
% coordinates the example prints, no two edges crossing. Build: make figures.
\documentclass[border=4pt]{standalone}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage{amsmath,amssymb}
\usepackage{tikz}
\usepackage{lamalunga}
\input{tikz-ablemon}
\begin{document}
\begin{tikzpicture}[abl,x=10mm,y=10mm]
  \foreach \i in {0,...,6} {
    \draw[guide,draw=ablInk!20] (\i,0) -- (\i,6);
    \draw[guide,draw=ablInk!20] (0,\i) -- (6,\i);
    \node[index] at (\i,-0.75) {\i};
    \node[index] at (-0.75,\i) {\i};
  }
  \foreach \n/\x/\y in {0/2/3,1/3/2,2/1/3,3/2/1,4/3/3,5/6/1,6/0/6,7/1/0}
    \node[vertex] (\n) at (\x,\y) {\n};
  \foreach \a/\b in {0/1,0/2,0/4,1/3,1/5,2/3,2/6,3/7,4/5,4/6,5/7,6/7}
    \draw[edge-touched] (\a) -- (\b);
\end{tikzpicture}
\end{document}
