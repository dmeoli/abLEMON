% The classes of the LP and MIP interface (sec. 20.1): LpBase holds the
% model, LpSolver and MipSolver add the solving and the solution, and each
% backend derives from one of the two (the common bases of the GLPK, CPLEX
% and skeleton classes are left out). Build: make figures (in ../build).
\documentclass[border=4pt]{standalone}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage{amsmath,amssymb}
\usepackage{tikz}
\usepackage{lamalunga}
\input{tikz-ablemon}
\tikzset{cls/.style={box,minimum width=20mm,font=\footnotesize\ttfamily},
         cls-touched/.style={box-touched,minimum width=20mm,
                             font=\footnotesize\ttfamily\bfseries},
         cls-added/.style={box-added,minimum width=20mm,
                           font=\footnotesize\ttfamily\bfseries},
         isa/.style={->,line width=0.9pt,draw=ablInk!55}}
\begin{document}
\begin{tikzpicture}[abl]
  \node[cls-touched] (base) at (75mm,0) {LpBase};
  \node[cls-touched] (lps) at (34mm,-16mm) {LpSolver};
  \node[cls-touched] (mips) at (118mm,-16mm) {MipSolver};
  \draw[isa] (lps.north) |- ++(0,4mm) -| (base.south west);
  \draw[isa] (mips.north) |- ++(0,4mm) -| (base.south east);
  \foreach \c/\x in {GlpkLp/0,CplexLp/16,ClpLp/32,SoplexLp/49,
                     LpSkeleton/68} {
    \node[cls-added,minimum width=11mm,inner sep=1.5pt] (\c)
      at (\x mm,-34mm) {\c};
    \draw[isa] (\c.north) -- ++(0,4mm) -| (lps.south);
  }
  \foreach \c/\x in {GlpkMip/94,CplexMip/111,CbcMip/127,MipSkeleton/145} {
    \node[cls-added,minimum width=11mm,inner sep=1.5pt] (\c)
      at (\x mm,-34mm) {\c};
    \draw[isa] (\c.north) -- ++(0,4mm) -| (mips.south);
  }
\end{tikzpicture}
\end{document}
