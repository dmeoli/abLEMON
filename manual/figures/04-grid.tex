% GridGraph(4, 3) (sec. 4.6): node (i, j) has the index i + 4j, the origin
% is at the bottom left, and right(), left(), up() and down() give the arcs
% out of node (1, 1). Build: make figures (in ../build).
\documentclass[border=4pt]{standalone}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage{amsmath,amssymb}
\usepackage{tikz}
\usepackage{lamalunga}
\input{tikz-ablemon}
\begin{document}
\begin{tikzpicture}[abl]
  \foreach \i in {0,...,3}
    \foreach \j in {0,...,2} {
      \pgfmathtruncatemacro{\id}{\i+4*\j}
      \node[vertex] (n\i\j) at (\i*26mm,\j*20mm) {\id};
    }
  \node[vertex-touched] at (n11) {5};
  \foreach \i in {0,...,3}
    \foreach \j/\k in {0/1,1/2}
      \draw[edge] (n\i\j) -- (n\i\k);
  \foreach \j in {0,...,2}
    \foreach \i/\k in {0/1,1/2,2/3}
      \draw[edge] (n\i\j) -- (n\k\j);
  \draw[arc-touched] (n11) -- node[label] {\ttfamily right} (n21);
  \draw[arc-touched] (n11) -- node[label] {\ttfamily left} (n01);
  \draw[arc-touched] (n11) -- node[label] {\ttfamily up} (n12);
  \draw[arc-touched] (n11) -- node[label] {\ttfamily down} (n10);
  \foreach \i in {0,...,3}
    \node[index,below=3mm of n\i0] {col \i};
  \foreach \j in {0,...,2}
    \node[index,left=3mm of n0\j] {row \j};
\end{tikzpicture}
\end{document}
